COPY TERMINAL OUTPUT (KEYBOARD-ONLY):
  focus source pane -> Ctrl+g s e -> scrollback opens in Neovim
  select lines (V, j/k) -> Space z c -> :q! -> switch destination
  in Neovim: Space z p; in terminal/chat input: Ctrl+g p v

COPY FROM NEOVIM INTO CHAT:
  Visual select code -> Space z c -> focus chat INPUT -> Ctrl+g p v
